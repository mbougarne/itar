cmake_minimum_required(VERSION 3.20)
project(itar_android LANGUAGES CXX)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(itar_package "${CMAKE_CURRENT_SOURCE_DIR}/.." REALPATH)
get_filename_component(itar_checkout "${itar_package}/../.." ABSOLUTE)
# npm pack stages a copy inside the package, which can outlive the source revision.
# A workspace build always follows the checkout, including through node_modules symlinks.
if(itar_package STREQUAL "${itar_checkout}/packages/itar" AND EXISTS "${itar_checkout}/core/include/itar/state.hpp")
  add_subdirectory("${itar_checkout}/core" "${CMAKE_CURRENT_BINARY_DIR}/core")
else()
  add_subdirectory("${itar_package}/cpp" "${CMAKE_CURRENT_BINARY_DIR}/core")
endif()
add_library(itar_android SHARED src/main/cpp/connection.cpp)
target_link_libraries(itar_android PRIVATE itar::core)
target_compile_options(itar_android PRIVATE -Wall -Wextra -Wpedantic -Werror)
set_target_properties(itar_android PROPERTIES CXX_EXTENSIONS OFF)
