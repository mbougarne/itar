cmake_minimum_required(VERSION 3.20)
project(itar_core LANGUAGES CXX)

add_library(itar_core STATIC
  src/build.cpp src/registry.cpp src/topology.cpp src/batch.cpp src/state.cpp
  src/host.cpp src/geometry.cpp src/focus.cpp src/navigation.cpp
  src/navigation_boundary.cpp src/navigation_order.cpp)
add_library(itar::core ALIAS itar_core)
target_compile_features(itar_core PUBLIC cxx_std_20)
target_include_directories(itar_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
set_target_properties(itar_core PROPERTIES CXX_EXTENSIONS OFF)

# Disposable probe behavior is separate from the production engine target.
add_library(itar_probe STATIC src/probe_surface.cpp)
add_library(itar::probe ALIAS itar_probe)
target_compile_features(itar_probe PUBLIC cxx_std_20)
target_include_directories(itar_probe PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
set_target_properties(itar_probe PROPERTIES CXX_EXTENSIONS OFF)

include(CTest)
if(BUILD_TESTING)
  function(itar_core_test name)
    add_executable(itar_${name}_tests tests/${name}.cpp)
    target_link_libraries(itar_${name}_tests PRIVATE itar::core)
    set_target_properties(itar_${name}_tests PROPERTIES CXX_EXTENSIONS OFF)
    add_test(NAME core.${name} COMMAND itar_${name}_tests)
  endfunction()
  itar_core_test(identity)
  itar_core_test(registry)
  itar_core_test(topology)
  itar_core_test(batch)
  itar_core_test(state)
  itar_core_test(host)
  itar_core_test(geometry)
  itar_core_test(focus)
  itar_core_test(navigation)
  itar_core_test(navigation_boundary)
  itar_core_test(navigation_order)
  add_executable(itar_apple_facade_tests ../probes/apple/tests/facade.cpp ../probes/apple/ItarProbe.cpp)
  target_include_directories(itar_apple_facade_tests PRIVATE ../probes/apple)
  target_link_libraries(itar_apple_facade_tests PRIVATE itar::probe)
  add_test(NAME probe.apple_facade COMMAND itar_apple_facade_tests)
  add_executable(itar_probe_tests tests/probe_surface.cpp)
  target_link_libraries(itar_probe_tests PRIVATE itar::probe)
  set_target_properties(itar_probe_tests PROPERTIES CXX_EXTENSIONS OFF)
  add_test(NAME probe.surface COMMAND itar_probe_tests)
  add_executable(itar_core_harness tests/harness.cpp)
  target_link_libraries(itar_core_harness PRIVATE itar::core)
  set_target_properties(itar_core_harness PROPERTIES CXX_EXTENSIONS OFF)
  add_test(NAME core.harness COMMAND itar_core_harness)
  add_test(NAME core.harness_failure COMMAND itar_core_harness --fail)
  set_tests_properties(core.harness_failure PROPERTIES WILL_FAIL TRUE)
endif()
