cmake_minimum_required(VERSION 3.20)
project(itar_core LANGUAGES CXX)

add_library(itar_core STATIC
  src/build.cpp src/registry.cpp src/topology.cpp src/batch.cpp src/state.cpp src/collection.cpp
  src/host.cpp src/geometry.cpp src/focus.cpp src/transaction_start.cpp src/transaction_reveal.cpp src/transaction_outcome.cpp src/transaction_cancel.cpp src/transaction_time.cpp src/transaction_input.cpp src/transaction_bounds.cpp src/navigation.cpp
  src/navigation_boundary.cpp src/navigation_order.cpp src/navigation_geometry.cpp
  src/navigation_entry.cpp src/navigation_recovery.cpp)
add_library(itar::core ALIAS itar_core)
target_compile_features(itar_core PUBLIC cxx_std_20)
target_include_directories(itar_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
set_target_properties(itar_core PROPERTIES CXX_EXTENSIONS OFF)

# C-compatible production boundary shared with the Swift CocoaPods target.
find_package(Threads REQUIRED)
add_library(itar_apple_engine STATIC ../packages/itar/apple/ItarEngine.cpp)
target_link_libraries(itar_apple_engine PUBLIC itar::core Threads::Threads)
target_include_directories(itar_apple_engine PUBLIC ../packages/itar/apple)
set_target_properties(itar_apple_engine PROPERTIES CXX_EXTENSIONS OFF)

# Disposable probe behavior is separate from the production engine target.
add_library(itar_probe STATIC src/probe_surface.cpp)
add_library(itar::probe ALIAS itar_probe)
target_compile_features(itar_probe PUBLIC cxx_std_20)
target_include_directories(itar_probe PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
set_target_properties(itar_probe PROPERTIES CXX_EXTENSIONS OFF)
